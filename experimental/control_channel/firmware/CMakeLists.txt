cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico CACHE STRING "Qualified target candidate: original RP2040 Pico")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(acvm_channel C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(acvm_channel pico.cpp)
target_link_libraries(acvm_channel pico_stdlib hardware_spi hardware_watchdog)
target_compile_options(acvm_channel PRIVATE -Wall -Wextra -Werror)
# Bound blocking USB writes; a stalled USB peer must not stop servicing the lease.
target_compile_definitions(acvm_channel PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=1000)
pico_enable_stdio_usb(acvm_channel 1)
pico_enable_stdio_uart(acvm_channel 0)
pico_add_extra_outputs(acvm_channel)
